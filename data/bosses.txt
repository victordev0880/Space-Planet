# --- Colmeia Lunar ------------------------------------------------------------
# Uma massa viva que se agarra à Lua. Bolsas pelo corpo dela incham (aviso) e
# soltam drones; estourada antes, a ninhada morre junto. Todas as bolsas
# estouradas, o coração (voltado para o planeta) abre. Da fase 2 em diante
# nascem propulsores atrás da Lua que a empurram contra a Terra; perto, um
# tentáculo arranca a crosta. Destruir os propulsores faz a Lua voltar.
# Bolsas estouradas deixam esporos: pegue com o raio trator e arremesse no coração.

hive.points         = 15000
hive.xp             = 40

hive.sac_hp       = 12        # vida de cada bolsa (são seis)
hive.sac_regrow   = 4         # segundos para as bolsas renascerem depois que o coração fecha
hive.inflate_time = 2.6       # aviso: a bolsa incha este tanto antes de soltar os drones

hive.inflate_3    = 3.8
hive.brood        = 3         # drones do Enxame por bolsa
hive.brood_cap    = 12        # com este tanto de drones em cena, a bolsa espera

hive.open_time  = 8           # segundos com o coração aberto
hive.core_armor = 0.05        # fração do dano que chega ao coração fechado
hive.core_open  = 0.3         # ... e na fase 3
hive.phase_2    = 0.65        # fração da vida em que começa a fase 2 (propulsores)
hive.phase_3    = 0.3         # ... e a fase 3 (cuspe e pressa)
